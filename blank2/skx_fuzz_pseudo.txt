FUNCTION skx_sad_decode(res):
  GET d0 = first skx_dev
  IF no d0:
    RETURN false
  SET d = d0
  GET addr = res.addr
  IF addr out of range (tohm or tolm gap):
    WARN
    RETURN false
  SET remote = 0
RESTART:
  SET prev_limit = 0
  FOR i in 0..SKX_MAX_SAD-1:
    GET sad from d, i
    IF fails:
      CONTINUE
    GET limit from sad
    IF enabled:
      IF addr in prev_limit to limit:
        GET ilv from d, i
        IF fails:
          RETURN false
        DETERMINE idx from interleave mode (bits 6-8, 8-10, 12-14, 30-32)
        GET tgt from ilv at 4*idx to 4*idx+3
        IF remote tgt:
          IF already remote:
            RETURN false
          SET remote = 1
          GET want = tgt target
          FIND d with imc[0].src_id == want
          IF found:
            SET d to that
            GOTO RESTART
          RETURN false
        DETERMINE lchan and shift based on mod3, mod3mode, mod3asmod2
        SET res.dev = d
        SET res.socket = d.imc[0].src_id
        IF no mcroute:
          RETURN false
        SET res.imc from mcroute bits for lchan
        SET res.channel from mcroute bits for lchan
        RETURN true
    SET prev_limit = limit + 1
  RETURN false

FUNCTION skx_tad_decode(res):
  FOR i in 0..SKX_MAX_TAD-1:
    GET base from res.dev, res.imc, i
    IF fails:
      CONTINUE
    GET wayness from res.dev, res.imc, i
    IF fails:
      CONTINUE
    IF addr in TAD_BASE to TAD_LIMIT:
      SET res.sktways from wayness
      SET res.chanways from wayness
      GET skt_interleave_bit from granularity[TAD_SKT_GRAN(base)]
      GET chn_interleave_bit from granularity[TAD_CHN_GRAN(base)]
      GET chnilvoffset from res.dev, res.imc, res.channel, i
      IF fails:
        RETURN false
      SET channel_addr = res.addr - TAD_OFFSET(chnilvoffset)
      IF chanways == 3 and skt > chn bit:
        INTERLEAVE channel_addr with chn bit, chanways, channel_addr
        INTERLEAVE channel_addr with skt bit, sktways, channel_addr
      ELSE:
        INTERLEAVE channel_addr with skt bit, sktways, res.addr
        INTERLEAVE channel_addr with chn bit, chanways, res.addr
      SET res.chan_addr = channel_addr
      RETURN true
  RETURN false

  FOR i in 0..nbits-1:

FUNCTION skx_bank_bits(addr, b0, b1, do_xor, x0, x1):
  SET ret = bit b0 of addr | (bit b1 of addr << 1)
  IF do_xor:
    SET xor_val = bit x0 of addr | (bit x1 of addr << 1)
    SET ret ^= xor_val
  RETURN ret

FUNCTION skx_mad_decode(r):
  GET dimm = r.dev.imc[r.imc].chan[r.channel].dimms[r.dimm]
  SET bg0 = 6 if dimm.fine_grain_bank else 13
  IF dimm.close_pg:
    SET r.row = skx_bits(r.rank_address, dimm.rowbits, skx_close_row)
    SET r.column = skx_bits(r.rank_address, dimm.colbits, skx_close_column)
    SET r.column |= 0x400
    SET r.bank_address = skx_bank_bits(r.rank_address, 8, 9, dimm.bank_xor_enable, 22, 28)
    SET r.bank_group = skx_bank_bits(r.rank_address, 6, 7, dimm.bank_xor_enable, 20, 21)
  ELSE:
    SET r.row = skx_bits(r.rank_address, dimm.rowbits, skx_open_row)
    IF dimm.fine_grain_bank:
      SET r.column = skx_bits(r.rank_address, dimm.colbits, skx_open_fine_column)
    ELSE:
      SET r.column = skx_bits(r.rank_address, dimm.colbits, skx_open_column)
    SET r.bank_address = skx_bank_bits(r.rank_address, 18, 19, dimm.bank_xor_enable, 22, 23)
    SET r.bank_group = skx_bank_bits(r.rank_address, bg0, 17, dimm.bank_xor_enable, 20, 21)
  MASK r.row to (1 << dimm.rowbits) - 1
  RETURN true
